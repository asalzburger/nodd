cmake_minimum_required(VERSION 3.20)
project(nODDLongStripPrototype LANGUAGES CXX)
find_package(DD4hep 1.37 REQUIRED CONFIG COMPONENTS DDCore)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(nODDLongStrip SHARED LongStrip.cpp)
target_compile_features(nODDLongStrip PRIVATE cxx_std_20)
target_link_libraries(nODDLongStrip PUBLIC DD4hep::DDCore)
dd4hep_generate_rootmap(nODDLongStrip)
get_filename_component(NODD_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NODD_COMPACT "${CMAKE_CURRENT_BINARY_DIR}/compact")
file(GLOB NODD_INPUTS "${NODD_REPO}/tools/long_strip_barrel/*.py" "${NODD_REPO}/tools/long_strip_barrel/*.json")
add_custom_command(OUTPUT "${NODD_COMPACT}/long-strip.xml" "${NODD_COMPACT}/expected.json"
 COMMAND "${Python3_EXECUTABLE}" -B "${NODD_REPO}/tools/long_strip_barrel/export.py" --output "${NODD_COMPACT}"
 DEPENDS ${NODD_INPUTS} "${NODD_REPO}/tools/pixel_barrel_dd4hep/materials.py" WORKING_DIRECTORY "${NODD_REPO}" VERBATIM)
add_custom_target(nodd-long-strip-compact ALL DEPENDS "${NODD_COMPACT}/long-strip.xml" "${NODD_COMPACT}/expected.json")
include(CTest)
if(BUILD_TESTING)
 add_test(NAME long-strip-model COMMAND "${Python3_EXECUTABLE}" -B -m unittest discover -s "${NODD_REPO}/tools/long_strip_barrel" -p "test_*.py" -v)
 add_test(NAME long-strip-native COMMAND "${Python3_EXECUTABLE}" -B "${NODD_REPO}/tools/long_strip_barrel/validate.py" --compact "${NODD_COMPACT}/long-strip.xml" --expected "${NODD_COMPACT}/expected.json" --library "$<TARGET_FILE:nODDLongStrip>" --output "${CMAKE_CURRENT_BINARY_DIR}/native.json")
 set_tests_properties(long-strip-native PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}:$ENV{LD_LIBRARY_PATH};DD4HEP_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}:$ENV{DD4HEP_LIBRARY_PATH};DYLD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}:$ENV{DYLD_LIBRARY_PATH}")
endif()
