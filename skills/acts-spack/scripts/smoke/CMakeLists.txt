cmake_minimum_required(VERSION 3.20)
project(nodd_spack_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(DD4hep REQUIRED COMPONENTS DDCore DDG4)
find_package(Geant4 REQUIRED)
add_executable(spack-smoke smoke.cpp)
target_include_directories(spack-smoke PRIVATE ${Geant4_INCLUDE_DIRS})
target_link_libraries(spack-smoke PRIVATE DD4hep::DDCore DD4hep::DDG4 ${Geant4_LIBRARIES})
