cmake_minimum_required(VERSION 3.24)
project(NoddTrackerOptimization LANGUAGES CXX)
set(NODD_ACTS_SOURCE "" CACHE PATH "Source matching existing ACTS runtime")
set(NODD_ACTS_BUILD "" CACHE PATH "Existing ACTS build; never modified")
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
find_package(Boost CONFIG REQUIRED)
find_library(ACTS_CORE ActsCore PATHS "${NODD_ACTS_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
pybind11_add_module(nodd_tracker_material MaterialTools.cpp)
target_compile_features(nodd_tracker_material PRIVATE cxx_std_20)
target_include_directories(nodd_tracker_material PRIVATE "${NODD_ACTS_SOURCE}/Core/include" "${NODD_ACTS_BUILD}/Core/include")
target_link_libraries(nodd_tracker_material PRIVATE "${ACTS_CORE}" Eigen3::Eigen Boost::headers)
set_target_properties(nodd_tracker_material PROPERTIES BUILD_RPATH "${NODD_ACTS_BUILD}/lib")
