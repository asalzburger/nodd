cmake_minimum_required(VERSION 3.24)
project(NoddTtbarSafeGbts LANGUAGES CXX)
set(NODD_ACTS_SOURCE "" CACHE PATH "Read-only exact ACTS355ea source")
set(NODD_ACTS_BUILD "" CACHE PATH "Read-only native ACTS build")
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 3.0.4 EXACT CONFIG REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
find_package(Boost CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
file(STRINGS "${NODD_ACTS_BUILD}/CMakeCache.txt" native_compiler REGEX "^CMAKE_CXX_COMPILER:FILEPATH=")
string(REPLACE "CMAKE_CXX_COMPILER:FILEPATH=" "" native_compiler "${native_compiler}")
file(REAL_PATH "${native_compiler}" native_real)
file(REAL_PATH "${CMAKE_CXX_COMPILER}" current_real)
if(NOT native_real STREQUAL current_real)
  message(FATAL_ERROR "Compiler differs from native ACTS Core")
endif()
file(GLOB local_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${local_sources} "${NODD_ACTS_BUILD}/build.ninja")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${Python_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
  "${NODD_ACTS_SOURCE}" "${NODD_ACTS_BUILD}" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}"
  COMMAND_ERROR_IS_FATAL ANY)
include("${CMAKE_CURRENT_BINARY_DIR}/generated/native.cmake")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${NODD_UPSTREAM_INPUTS})
separate_arguments(native_flags UNIX_COMMAND "${NODD_NATIVE_FLAGS}")
separate_arguments(native_defines UNIX_COMMAND "${NODD_NATIVE_DEFINES}")
foreach(component Core ExamplesFramework PluginFpeMonitoring)
  find_library(ACTS_${component} Acts${component} PATHS "${NODD_ACTS_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
file(GLOB generated_sources "${CMAKE_CURRENT_BINARY_DIR}/generated/*.cpp")
pybind11_add_module(nodd_ttbar_gbts NO_EXTRAS Binding.cpp ${generated_sources})
target_compile_features(nodd_ttbar_gbts PRIVATE cxx_std_20)
target_compile_options(nodd_ttbar_gbts PRIVATE ${native_flags} ${native_defines})
target_compile_definitions(nodd_ttbar_gbts PRIVATE NODD_EXPECTED_PYBIND11_INTERNALS_VERSION=11)
target_include_directories(nodd_ttbar_gbts BEFORE PRIVATE ${pybind11_INCLUDE_DIRS})
target_include_directories(nodd_ttbar_gbts PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated"
  "${NODD_ACTS_SOURCE}/Core/include" "${NODD_ACTS_BUILD}/Core/include"
  "${NODD_ACTS_SOURCE}/Examples/Framework/include" "${NODD_ACTS_SOURCE}/Plugins/FpeMonitoring/include")
target_link_libraries(nodd_ttbar_gbts PRIVATE "${ACTS_Core}" "${ACTS_ExamplesFramework}" "${ACTS_PluginFpeMonitoring}"
  Eigen3::Eigen Boost::headers nlohmann_json::nlohmann_json)
set_target_properties(nodd_ttbar_gbts PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON BUILD_RPATH "${NODD_ACTS_BUILD}/lib")

# Both controls compile the actual generated filter/geometry sources. Only
# test access friendship is enabled; native and strict targets are separate.
enable_testing()
set(numeric_sources test_lencorr.cpp)
foreach(variant Renamed Guarded)
  list(APPEND numeric_sources "${CMAKE_CURRENT_BINARY_DIR}/generated/Nodd${variant}GbtsTrackingFilter.cpp"
    "${CMAKE_CURRENT_BINARY_DIR}/generated/Nodd${variant}GbtsGeometry.cpp")
endforeach()
foreach(test_target test_gbts_lencorr test_gbts_lencorr_strict)
  add_executable(${test_target} ${numeric_sources})
  target_compile_features(${test_target} PRIVATE cxx_std_20)
  target_compile_options(${test_target} PRIVATE ${native_flags} ${native_defines})
  target_compile_definitions(${test_target} PRIVATE NODD_GBTS_TEST_ACCESS=1)
  target_include_directories(${test_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${NODD_ACTS_SOURCE}/Core/include" "${NODD_ACTS_BUILD}/Core/include" "${NODD_ACTS_SOURCE}/Plugins/FpeMonitoring/include")
  target_link_libraries(${test_target} PRIVATE "${ACTS_Core}" "${ACTS_PluginFpeMonitoring}" Eigen3::Eigen Boost::headers nlohmann_json::nlohmann_json)
  set_target_properties(${test_target} PROPERTIES BUILD_RPATH "${NODD_ACTS_BUILD}/lib")
endforeach()
target_compile_options(test_gbts_lencorr_strict PRIVATE -ffp-model=strict)
target_compile_definitions(test_gbts_lencorr_strict PRIVATE NODD_STRICT_FIXTURE=1)
add_test(NAME gbts_lencorr_native COMMAND test_gbts_lencorr)
add_test(NAME gbts_lencorr_strict COMMAND test_gbts_lencorr_strict)
