cmake_minimum_required(VERSION 3.24)
project(NoddTrackerReconstruction LANGUAGES CXX)
set(NODD_ACTS_SOURCE "" CACHE PATH "Matching installed ACTS source")
set(NODD_ACTS_BUILD "" CACHE PATH "Existing read-only ACTS build")
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 3.0.4 EXACT CONFIG REQUIRED)
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${NODD_ACTS_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE NODD_ACTS_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT NODD_ACTS_REVISION STREQUAL "355ea68493b326956756c9386d2fd9eaf9328568")
  message(FATAL_ERROR "Reconstruction observer requires source matching installed ACTS355ea684; got ${NODD_ACTS_REVISION}")
endif()
find_package(Eigen3 CONFIG REQUIRED)
find_package(Boost CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Geant4 REQUIRED)
find_package(DD4hep CONFIG REQUIRED)
find_library(ACTS_DETECTOR ActsExamplesDetectorsCommon PATHS "${NODD_ACTS_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(ACTS_DD4HEP ActsExamplesDetectorDD4hep PATHS "${NODD_ACTS_BUILD}/Examples/Detectors/DD4hepDetector" "${NODD_ACTS_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(ACTS_GEANT4 ActsExamplesGeant4 PATHS "${NODD_ACTS_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(ACTS_CORE ActsCore PATHS "${NODD_ACTS_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(ACTS_FRAMEWORK ActsExamplesFramework PATHS "${NODD_ACTS_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
pybind11_add_module(nodd_reconstruction Reconstruction.cpp)
target_compile_features(nodd_reconstruction PRIVATE cxx_std_20)
target_compile_definitions(nodd_reconstruction PRIVATE NODD_EXPECTED_PYBIND11_INTERNALS_VERSION=11)
target_include_directories(nodd_reconstruction PRIVATE "${NODD_ACTS_SOURCE}/Core/include"
  "${NODD_ACTS_SOURCE}/Fatras/include" "${NODD_ACTS_SOURCE}/Examples/Framework/include"
  "${NODD_ACTS_SOURCE}/Plugins/FpeMonitoring/include" "${NODD_ACTS_BUILD}/Core/include")
target_include_directories(nodd_reconstruction PRIVATE "${NODD_ACTS_SOURCE}/Examples/Algorithms/Geant4/include" ${Geant4_INCLUDE_DIRS})
target_include_directories(nodd_reconstruction PRIVATE "${NODD_ACTS_SOURCE}/Examples/Detectors/Common/include")
target_include_directories(nodd_reconstruction PRIVATE "${NODD_ACTS_SOURCE}/Examples/Detectors/DD4hepDetector/include" "${NODD_ACTS_SOURCE}/Plugins/DD4hep/include")
target_include_directories(nodd_reconstruction PRIVATE "${NODD_ACTS_SOURCE}/Plugins/Root/include")
target_link_libraries(nodd_reconstruction PRIVATE "${ACTS_CORE}" "${ACTS_FRAMEWORK}"
  Eigen3::Eigen Boost::headers nlohmann_json::nlohmann_json)
target_link_libraries(nodd_reconstruction PRIVATE "${ACTS_GEANT4}" ${Geant4_LIBRARIES})
target_link_libraries(nodd_reconstruction PRIVATE "${ACTS_DETECTOR}" DD4hep::DDCore)
target_link_libraries(nodd_reconstruction PRIVATE "${ACTS_DD4HEP}")
set_target_properties(nodd_reconstruction PROPERTIES CXX_VISIBILITY_PRESET default
  VISIBILITY_INLINES_HIDDEN OFF BUILD_RPATH "${NODD_ACTS_BUILD}/lib")
