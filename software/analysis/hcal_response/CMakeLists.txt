cmake_minimum_required(VERSION 3.20)
project(nODDHCALResponseAudit LANGUAGES CXX)
find_package(DD4hep 1.38 REQUIRED CONFIG COMPONENTS DDG4)
find_package(Geant4 REQUIRED)
add_library(nODDHCALResponseAudit SHARED Audit.cpp)
target_compile_features(nODDHCALResponseAudit PRIVATE cxx_std_17)
target_include_directories(nODDHCALResponseAudit PRIVATE ${Geant4_INCLUDE_DIRS})
target_link_libraries(nODDHCALResponseAudit PUBLIC DD4hep::DDG4 ${Geant4_LIBRARIES})
dd4hep_generate_rootmap(nODDHCALResponseAudit)
