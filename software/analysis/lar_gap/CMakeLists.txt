cmake_minimum_required(VERSION 3.20)
project(nODDLArStepObserver LANGUAGES CXX)
find_package(DD4hep 1.38 REQUIRED CONFIG COMPONENTS DDG4)
find_package(Geant4 REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(nODDLArStepObserver SHARED StepAudit.cpp)
target_compile_features(nODDLArStepObserver PRIVATE cxx_std_17)
target_include_directories(nODDLArStepObserver PRIVATE ${Geant4_INCLUDE_DIRS})
target_link_libraries(nODDLArStepObserver PUBLIC DD4hep::DDG4 ${Geant4_LIBRARIES} PRIVATE nlohmann_json::nlohmann_json)
dd4hep_generate_rootmap(nODDLArStepObserver)
