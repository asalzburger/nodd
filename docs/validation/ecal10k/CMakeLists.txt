cmake_minimum_required(VERSION 3.20)
project(nODDAngularAudit LANGUAGES CXX)
find_package(DD4hep 1.38 REQUIRED CONFIG COMPONENTS DDG4)
find_package(Geant4 REQUIRED)
add_library(nODDAngularAudit SHARED Audit.cpp)
target_compile_features(nODDAngularAudit PRIVATE cxx_std_17)
target_include_directories(nODDAngularAudit PRIVATE ${Geant4_INCLUDE_DIRS})
target_link_libraries(nODDAngularAudit PUBLIC DD4hep::DDG4 ${Geant4_LIBRARIES})
dd4hep_generate_rootmap(nODDAngularAudit)
